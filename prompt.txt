
CREATE DATABASE IF NOT EXISTS transolutions_vmms
  CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE transolutions_vmms;

SET FOREIGN_KEY_CHECKS = 0;

-- ---------------------------------------------------------------------
-- 1. USERS AND ROLES
-- ---------------------------------------------------------------------
CREATE TABLE roles (
  id          TINYINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  name        VARCHAR(30)  NOT NULL UNIQUE,   -- admin, secretary, purchaser, mechanic, customer
  description VARCHAR(150) NULL
) ENGINE=InnoDB;

CREATE TABLE users (
  id                BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  role_id           TINYINT UNSIGNED NOT NULL,
  name              VARCHAR(120) NOT NULL,
  email             VARCHAR(150) NOT NULL UNIQUE,
  phone             VARCHAR(20)  NULL,
  password          VARCHAR(255) NOT NULL,
  is_active         TINYINT(1)   NOT NULL DEFAULT 1,
  email_verified_at TIMESTAMP NULL,
  remember_token    VARCHAR(100) NULL,
  last_login_at     TIMESTAMP NULL,
  created_at        TIMESTAMP NULL,
  updated_at        TIMESTAMP NULL,
  deleted_at        TIMESTAMP NULL,
  CONSTRAINT fk_users_role FOREIGN KEY (role_id) REFERENCES roles(id),
  INDEX idx_users_role (role_id)
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- 2. CUSTOMERS, AFFILIATES, VEHICLES
-- ---------------------------------------------------------------------
CREATE TABLE affiliates (
  id             BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  name           VARCHAR(150) NOT NULL,
  contact_person VARCHAR(120) NULL,
  phone          VARCHAR(20)  NULL,
  address        VARCHAR(255) NULL,
  is_active      TINYINT(1) NOT NULL DEFAULT 1,
  created_at     TIMESTAMP NULL,
  updated_at     TIMESTAMP NULL
) ENGINE=InnoDB;

CREATE TABLE customers (
  id           BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  user_id      BIGINT UNSIGNED NULL UNIQUE,           -- set when customer has a portal account
  affiliate_id BIGINT UNSIGNED NULL,
  first_name   VARCHAR(80)  NOT NULL,
  last_name    VARCHAR(80)  NOT NULL,
  phone        VARCHAR(20)  NOT NULL,
  email        VARCHAR(150) NULL,
  address      VARCHAR(255) NULL,
  referral_source ENUM('walk_in','referral','phone','messenger','affiliate','other')
                 NOT NULL DEFAULT 'walk_in',
  notes        TEXT NULL,
  created_at   TIMESTAMP NULL,
  updated_at   TIMESTAMP NULL,
  deleted_at   TIMESTAMP NULL,
  CONSTRAINT fk_cust_user      FOREIGN KEY (user_id)      REFERENCES users(id)      ON DELETE SET NULL,
  CONSTRAINT fk_cust_affiliate FOREIGN KEY (affiliate_id) REFERENCES affiliates(id) ON DELETE SET NULL,
  INDEX idx_cust_name (last_name, first_name),
  INDEX idx_cust_phone (phone)
) ENGINE=InnoDB;

CREATE TABLE vehicles (
  id            BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  customer_id   BIGINT UNSIGNED NOT NULL,
  plate_number  VARCHAR(15) NOT NULL UNIQUE,
  make          VARCHAR(60) NOT NULL,                  -- Chinese / European makes mostly
  model         VARCHAR(60) NOT NULL,
  year          SMALLINT UNSIGNED NULL,
  color         VARCHAR(30) NULL,
  transmission  ENUM('automatic','manual','cvt','dct','other') NOT NULL DEFAULT 'automatic',
  engine_type   VARCHAR(60) NULL,
  vin           VARCHAR(30) NULL,
  odometer_km   INT UNSIGNED NULL,
  created_at    TIMESTAMP NULL,
  updated_at    TIMESTAMP NULL,
  deleted_at    TIMESTAMP NULL,
  CONSTRAINT fk_veh_customer FOREIGN KEY (customer_id) REFERENCES customers(id),
  INDEX idx_veh_customer (customer_id)
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- 3. SERVICE CATALOG
-- ---------------------------------------------------------------------
CREATE TABLE services (
  id               BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  name             VARCHAR(120) NOT NULL UNIQUE,
  category         ENUM('transmission','engine_troubleshooting','general_maintenance','detailing') NOT NULL,
  maintenance_type ENUM('preventive','routine','corrective','emergency') NOT NULL,
  description      TEXT NULL,
  base_price_min   DECIMAL(10,2) NULL,                 -- prices vary per diagnosis
  base_price_max   DECIMAL(10,2) NULL,
  est_hours        DECIMAL(5,1) NULL,
  is_complex       TINYINT(1) NOT NULL DEFAULT 0,      -- complex jobs are done at night
  is_active        TINYINT(1) NOT NULL DEFAULT 1,
  created_at       TIMESTAMP NULL,
  updated_at       TIMESTAMP NULL
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- 4. INQUIRIES (Messenger / phone / walk-in / social media)
-- ---------------------------------------------------------------------
CREATE TABLE inquiries (
  id            BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  customer_id   BIGINT UNSIGNED NULL,
  contact_name  VARCHAR(120) NULL,
  contact_info  VARCHAR(150) NULL,
  channel       ENUM('messenger','phone','walk_in','referral','social_media','other') NOT NULL,
  message       TEXT NOT NULL,
  status        ENUM('new','in_progress','converted','closed') NOT NULL DEFAULT 'new',
  handled_by    BIGINT UNSIGNED NULL,
  booking_id    BIGINT UNSIGNED NULL,                  -- filled on "Convert to booking"
  created_at    TIMESTAMP NULL,
  updated_at    TIMESTAMP NULL,
  CONSTRAINT fk_inq_customer FOREIGN KEY (customer_id) REFERENCES customers(id) ON DELETE SET NULL,
  CONSTRAINT fk_inq_handler  FOREIGN KEY (handled_by)  REFERENCES users(id)     ON DELETE SET NULL,
  INDEX idx_inq_status (status)
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- 5. BOOKINGS AND SERVICE TRACKING
-- ---------------------------------------------------------------------
CREATE TABLE bookings (
  id               BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  booking_no       VARCHAR(20) NOT NULL UNIQUE,        -- e.g. TR-20261005-001
  customer_id      BIGINT UNSIGNED NOT NULL,
  vehicle_id       BIGINT UNSIGNED NOT NULL,
  source           ENUM('walk_in','referral','phone','messenger','affiliate','portal') NOT NULL,
  problem_description TEXT NULL,
  scheduled_at     DATETIME NULL,
  service_location ENUM('shop','pickup','on_site') NOT NULL DEFAULT 'shop',
  location_address VARCHAR(255) NULL,
  is_urgent        TINYINT(1) NOT NULL DEFAULT 0,
  status           ENUM('pending','scheduled','processing','completed','paid_released')
                     NOT NULL DEFAULT 'pending',
  created_by       BIGINT UNSIGNED NULL,
  created_at       TIMESTAMP NULL,
  updated_at       TIMESTAMP NULL,
  deleted_at       TIMESTAMP NULL,
  CONSTRAINT fk_book_customer FOREIGN KEY (customer_id) REFERENCES customers(id),
  CONSTRAINT fk_book_vehicle  FOREIGN KEY (vehicle_id)  REFERENCES vehicles(id),
  CONSTRAINT fk_book_creator  FOREIGN KEY (created_by)  REFERENCES users(id) ON DELETE SET NULL,
  INDEX idx_book_status (status),
  INDEX idx_book_schedule (scheduled_at),
  INDEX idx_book_customer (customer_id)
) ENGINE=InnoDB;

ALTER TABLE inquiries
  ADD CONSTRAINT fk_inq_booking FOREIGN KEY (booking_id) REFERENCES bookings(id) ON DELETE SET NULL;

CREATE TABLE booking_services (
  id          BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,

CREATE TABLE part_categories (

  id             BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  sku            VARCHAR(40)  NOT NULL UNIQUE,
  name           VARCHAR(150) NOT NULL,

  supplier_id    BIGINT UNSIGNED NULL,

-- Parts used per repair job (fixes the "parts used are not recorded" problem)
CREATE TABLE job_parts (
  id           BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  job_card_id  BIGINT UNSIGNED NOT NULL,
  part_id      BIGINT UNSIGNED NOT NULL,
  qty          INT NOT NULL,

  recorded_by  BIGINT UNSIGNED NOT NULL,               -- mechanic
  created_at   TIMESTAMP NULL,

-- Every stock change is logged: delivery in, job use out, adjustment
CREATE TABLE stock_movements (
  id             BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  part_id        BIGINT UNSIGNED NOT NULL,
  movement_type  ENUM('delivery_in','job_out','adjustment_in','adjustment_out','return_in') NOT NULL,
  qty            INT NOT NULL,                         -- always positive; type gives direction
  reference_type VARCHAR(40) NULL,                     -- goods_receipt / job_part / manual
  reference_id   BIGINT UNSIGNED NULL,
  balance_after  INT NOT NULL,
  remarks        VARCHAR(255) NULL,
  created_by     BIGINT UNSIGNED NULL,
  created_at     TIMESTAMP NULL,
  CONSTRAINT fk_sm_part FOREIGN KEY (part_id)    REFERENCES parts(id),
  CONSTRAINT fk_sm_user FOREIGN KEY (created_by) REFERENCES users(id) ON DELETE SET NULL,
  INDEX idx_sm_part_date (part_id, created_at)
) ENGINE=InnoDB;

-- ---------------------------------------------------------------------
-- 9. TRIGGERS: keep stock accurate automatically
-- ---------------------------------------------------------------------
DELIMITER $$

-- Parts used on a job are deducted from stock and logged
CREATE TRIGGER trg_job_parts_after_insert
AFTER INSERT ON job_parts
FOR EACH ROW
BEGIN
  UPDATE parts SET stock_qty = stock_qty - NEW.qty WHERE id = NEW.part_id;
  INSERT INTO stock_movements
    (part_id, movement_type, qty, reference_type, reference_id, balance_after, created_by, created_at)
  SELECT NEW.part_id, 'job_out', NEW.qty, 'job_part', NEW.id, p.stock_qty, NEW.recorded_by, NOW()
  FROM parts p WHERE p.id = NEW.part_id;
END$$

-- Received delivery items are added to stock and logged
CREATE TRIGGER trg_gr_items_after_insert
AFTER INSERT ON goods_receipt_items
FOR EACH ROW
BEGIN
  DECLARE good_qty INT;
  SET good_qty = NEW.qty_received - NEW.qty_damaged;
  UPDATE parts SET stock_qty = stock_qty + good_qty WHERE id = NEW.part_id;
  INSERT INTO stock_movements
    (part_id, movement_type, qty, reference_type, reference_id, balance_after, created_at)
  SELECT NEW.part_id, 'delivery_in', good_qty, 'goods_receipt', NEW.goods_receipt_id, p.stock_qty, NOW()
  FROM parts p WHERE p.id = NEW.part_id;
END$$

DELIMITER ;
